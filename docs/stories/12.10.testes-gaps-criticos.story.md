class TestBboxToStyle:
    def test_a4_page_conversion(self):
        style = _bbox_to_style([100, 200, 300, 250], page_width=595, page_height=842)
        assert "position:absolute" in style
        assert "left:133.4px" in style  # 100 * (794/595)
        assert "top:504.3px" in style   # (842-250) * (1123/842) = y convertido

    def test_custom_page_size(self):
        # Página não-A4
        style = _bbox_to_style([0, 0, 400, 200], page_width=400, page_height=600)
        assert "left:0.0px" in style
        assert "width:794.0px" in style  # 400 * (794/400) = 794

    def test_bbox_none_returns_empty(self):
        style = _bbox_to_style(None)
        assert style == "" or style is None

class TestCoverageCalculation:
    def test_coverage_without_field_tree(self):
        mappings = [{"xsd_field_path": "a.b", "pdf_text": "val"}]
        result = _calculate_coverage(mappings, field_tree=None)
        assert result["fields"]["total"] >= 0  # Não levanta exception
        # Com modo heurístico: total = len(mappings) ou blocos detectados

    def test_coverage_with_empty_flat_paths(self):
        result = _calculate_coverage([], field_tree={"flat_paths": []})
        assert result["fields"]["percentage"] == 0
        # Não deve ser NaN ou None

    def test_coverage_correct_calculation(self):
        flat_paths = ["a.b", "a.c", "a.d"]
        mappings = [
            {"xsd_field_path": "a.b"},
            {"xsd_field_path": "a.c"},
        ]
        result = _calculate_coverage(mappings, {"flat_paths": flat_paths})
        assert result["fields"]["mapped"] == 2
        assert result["fields"]["total"] == 3
        assert result["fields"].get("percentage", 67) == 67

describe('loadPipelineFields', () => {
