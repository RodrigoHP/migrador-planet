## Método

```python
# Verificação automática do output
def verify_e2e_output(result: dict, ground_truth: dict) -> dict:
    checks = {}
    
    # Stage 3: repeated sections
    tree = result.get("document_trees", [{}])[0]
    repeated = [n for n in tree.get("sections", []) if n.get("type") == "repeated_section"]
    checks["repeated_sections_found"] = len(repeated)
    checks["repeated_sections_expected"] = len(ground_truth["repeated_sections"])
    
    # Stage 4: list binding
    field_mappings = result.get("field_mappings", [])
    list_bindings = [m for m in field_mappings if m.get("binding_type") == "list"]
    checks["list_bindings_found"] = len(list_bindings)
    
    # Stage 5: repeat element in HTML
    html = result.get("template_draft", {}).get("html", "")
    checks["repeat_element_present"] = "<repeat" in html
    checks["repeat_data_list"] = bool(re.search(r'data-list="[^"]+\[\]"', html))
    
    return checks
```
