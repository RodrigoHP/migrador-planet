#### Fix 3 — Coverage deve diferenciar "0 mapeados" de "não pôde calcular"

```python
# backend/services/stages/template_draft.py
def _calculate_coverage(field_mappings, field_tree):
    if not field_tree or not field_tree.get("flat_paths"):
        return {
            "fields": {"mapped": 0, "total": 0},
            "error": "field_tree_missing"  # ← Frontend exibe aviso específico
        }

    flat_paths = [p for p in field_tree["flat_paths"] if p]
    total = len(flat_paths)
    mapped_paths = {
        m["xsd_field_path"]
        for m in field_mappings
        if m.get("xsd_field_path") and m["xsd_field_path"].strip()
    }
    mapped = len(mapped_paths & set(flat_paths))
    return {
        "fields": {"mapped": mapped, "total": total},
        "percentage": round(mapped / total * 100) if total > 0 else 0
    }
```
